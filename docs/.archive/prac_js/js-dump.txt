console.log(rightFuncs[2]());  // Expected: 2 (CORRECT!)

// Event delegation: listen on parent, handle on children

});

  newItem.classList.add('item');
  newItem.innerHTML = `Item ${itemCount}`;
  list.appendChild(newItem);
  console.log('Added new item');
});

// TEST THIS
// 1. Click Item 1, Item 2, Item 3 → See output
// 2. Click "Add Item" button → New item appears
// 3. Click new item → Event handler works (event delegation!)
